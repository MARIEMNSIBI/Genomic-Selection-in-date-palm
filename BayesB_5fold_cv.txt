library(readxl)
library(BGLR)
library(data.table)

# Phenotypic data
pheno_raw <- read_excel("Pheno_BLUEs_136.xlsx")

traits <- c("FL", "R", "G", "B", "L", "a", "b", "RAB", "Hue", "ANT", "MO", "SUC", "GLU", "FRU", "RSGFS", "OXA", "CIT", "TAR", "MAL", "SUI", "FUM")

pheno_raw[traits] <- lapply(pheno_raw[traits], function(x) as.numeric(as.character(x)))

Pheno <- as.matrix(pheno_raw[, -1])
rownames(Pheno) <- pheno_raw$Variety

# Genotypic data
geno_raw <- fread("SNPs_sample_filtered_INPUT.csv")
individuals <- geno_raw[[1]]

geno_mat <- as.matrix(geno_raw[, -1])
rownames(geno_mat) <- individuals

# SNP imputation
geno_imputed <- apply(geno_mat, 2, function(x) {
  x[is.na(x)] <- mean(x, na.rm = TRUE)
  return(x)
})

# Align phenotype and genotype individuals
common_ind <- intersect(rownames(Pheno), rownames(geno_imputed))
Pheno <- Pheno[common_ind, ]
X_geno <- geno_imputed[common_ind, ]

traits <- colnames(Pheno)[1:10] 

# Cross-validation parameters
cycles <- 50
num_folds <- 5

# Cross-validation loop across traits
for (trait in traits) {
  # Phenotype vector
  y_full <- Pheno[, trait]
  
  accuracy_cycles <- matrix(NA, nrow = cycles, ncol = 1)
  mse_cycles      <- matrix(NA, nrow = cycles, ncol = 1)
  bias_cycles     <- matrix(NA, nrow = cycles, ncol = 1)
  rel_mse_cycles  <- matrix(NA, nrow = cycles, ncol = 1)
  bias_std_cycles <- matrix(NA, nrow = cycles, ncol = 1)
  mse_std_cycles  <- matrix(NA, nrow = cycles, ncol = 1)
  
  runtime <- system.time({
    for (r in 1:cycles) {
      folds <- sample(1:num_folds, size = length(y_full), replace = TRUE)
      fold_accuracy <- numeric(num_folds)
      fold_mse      <- numeric(num_folds)
      fold_bias     <- numeric(num_folds)
      fold_rel_mse  <- numeric(num_folds)
      fold_bias_std <- numeric(num_folds)
      fold_mse_std  <- numeric(num_folds)
      
      for (fold in 1:num_folds) {
        train_idx <- which(folds != fold)
        test_idx  <- which(folds == fold)
        
        # Mask test phenotypes
        y_train <- y_full
        y_train[test_idx] <- NA
        
        # BayesB linear predictor
        ETA <- list(list(X = X_geno, model = 'BayesB'))
        # Fit BGLR model 
        fm <- BGLR(y = y_train, ETA = ETA, verbose = FALSE)
        
        # Predicted values for the test set
        y_hat_test <- fm$yHat[test_idx]
        y_true_test <- y_full[test_idx]
        
        # Fold variances
        var_y_test <- var(y_true_test, na.rm = TRUE)
        sd_y_test  <- sd(y_true_test, na.rm = TRUE)
        
        # Validation metrics
        fold_accuracy[fold] <- cor(y_hat_test, y_true_test, use = "complete.obs")
        fold_mse[fold]      <- mean((y_hat_test - y_true_test)^2, na.rm = TRUE)
        fold_bias[fold]     <- mean(y_hat_test - y_true_test, na.rm = TRUE)
        fold_rel_mse[fold]  <- fold_mse[fold] / var_y_test
        fold_bias_std[fold] <- fold_bias[fold] / sd_y_test
        fold_mse_std[fold]  <- fold_mse[fold] / sd_y_test
      }
      
      accuracy_cycles[r, 1] <- mean(fold_accuracy, na.rm = TRUE)
      mse_cycles[r, 1]      <- mean(fold_mse, na.rm = TRUE)
      bias_cycles[r, 1]     <- mean(fold_bias, na.rm = TRUE)
      rel_mse_cycles[r, 1]  <- mean(fold_rel_mse, na.rm = TRUE)
      bias_std_cycles[r, 1] <- mean(fold_bias_std, na.rm = TRUE)
      mse_std_cycles[r, 1]  <- mean(fold_mse_std, na.rm = TRUE)
    }
  })
  
write.table(accuracy_fold, "BayesB_5fold_cv_accuracy.txt", sep=" ", quote=F, col.names=F, row.names=T)
write.table(mse_fold, "BayesB_5fold_cv_mse.txt", sep=" ", quote=F, col.names=F, row.names=T)
write.table(bias_fold, "BayesB_5fold_cv_bias.txt", sep=" ", quote=F, col.names=F, row.names=T)
write.table(rel_mse, "BayesB_5fold_cv_rel_mse.txt", sep=" ", quote=F, col.names=F, row.names=T)
write.table(bias_std, "BayesB_5fold_cv_bias_std.txt", sep=" ", quote=F, col.names=F, row.names=T)
}
